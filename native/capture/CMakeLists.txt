cmake_minimum_required(VERSION 3.28)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(g1capture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "g1capture requires Windows x64 and MSVC 2022")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use the x64 generator")
endif()
find_package(JNI REQUIRED COMPONENTS JVM)
add_library(g1capture SHARED capture.cpp)
target_compile_features(g1capture PRIVATE cxx_std_20)
target_include_directories(g1capture PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_definitions(g1capture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(g1capture PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(g1capture PRIVATE d3d11 dxgi windowsapp dwmapi)
